\section{Research Experience}
\cventry{Jan 2026--Present}{Evidence-Use Auditing for Deep ECG Models}{First Author}{}{}{\begin{itemize}
  \item Addressed whether high-performing ECG classifiers actually depend on clinically relevant signal evidence rather than task-correlated shortcuts.
  \item Developed a matched-control perturbation framework that compares clinically targeted interventions against matched control perturbations across evidence windows and perturbation doses.
  \item Evaluated the framework on PTB-XL and CODE-15\% across model architectures and frozen-weight cross-dataset transfer; manuscript under review at \textit{IEEE Journal of Biomedical and Health Informatics} (JBHI).
\end{itemize}}

\cventry{Jun 2026--Present}{DLEF: Dataset-Label Evidence Fingerprints}{First Author}{}{}{\begin{itemize}
  \item Investigated whether dataset-specific evidence structure can explain why ECG classifiers lose performance when transferred across datasets.
  \item Proposed directional and boundary-aware evidence fingerprints and designed analyses relating these fingerprints to cross-dataset AUROC degradation.
  \item Evaluated CODE-15\%, PTB-XL, and Chapman-Shaoxing; manuscript under review at IEEE BIBM 2026.
\end{itemize}}

\cventry{Jul 2026--Present}{PathCLIP: Evidence-Aware ECG--Text Representation Learning}{Researcher}{}{}{\begin{itemize}
  \item Investigating how diagnostic knowledge and ECG findings can provide clinically meaningful supervision for ECG--text alignment and zero-shot recognition.
  \item Designed matched-count, permuted-hierarchy, residual-only, and held-out-label controls to distinguish genuine knowledge gains from additional supervision or representation capacity.
  \item Current work shifts from ontology-only supervision toward finding-level evidence alignment.
\end{itemize}}
