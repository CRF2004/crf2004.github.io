\section{Research Experience}
\noindent\textit{\hospitalaffiliation}\par
\researchentry{Jun 2026--Present}{DLEF: Explaining Why ECG Models Fail Across Hospitals}{First author; submitted to \textbf{IEEE BIBM 2026}}{\begin{itemize}
  \item \textbf{Problem:} Aggregate performance scores show that a model degrades at a new hospital, but do not identify which diagnostic labels or evidence distributions drive the failure.
  \item \textbf{Contribution:} Formulated the question and led study design, implementation, experiments, analysis, and writing; developed directional dataset-label ``evidence fingerprints.''
  \item \textbf{Finding:} Greater fingerprint mismatch predicted larger external AUROC degradation (Spearman $r=0.843$ across 108 comparisons; $r=0.892$ after aggregation), enabling label-level transfer-risk screening.
\end{itemize}}

\researchentry{Jan 2026--Present}{Matched-Control Perturbation Auditing: Does an ECG Model Use Clinical Evidence?}{First author; submitted to \textbf{IEEE JBHI}}{\begin{itemize}
  \item \textbf{Problem:} Accurate models may rely on spurious cues, while target-only perturbations can confuse generic signal damage with genuine use of clinical evidence.
  \item \textbf{Contribution:} Developed and led a matched-control audit covering clinically defined evidence units, perturbation operators, and frozen-weight external validation.
  \item \textbf{Finding:} Only 6 of 15 model-diagnosis pairs were evidence-supported; matched controls or operator choice changed several conclusions, showing that strong external performance alone does not establish clinically intended evidence use.
\end{itemize}}
